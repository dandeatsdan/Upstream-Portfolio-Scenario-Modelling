def melt_scenario_config(df):
    scenario_prefixes = sorted({
        col.split("_")[-1]
        for col in df.columns
        if col.startswith(("S", "Op_S", "P1_S", "P2_S", "P3_S", "P4_S", "P5_S"))
    })

    scenario_rows = []

    for prefix in scenario_prefixes:
        include_col = f"S{prefix}" if not prefix.startswith("S") else prefix
        op_col = f"Op_{include_col}"
        param1_col = f"P1_{include_col}"
        param2_col = f"P2_{include_col}"
        param3_col = f"P3_{include_col}"
        param4_col = f"P4_{include_col}"
        param5_col = f"P5_{include_col}"

        for col in [include_col, op_col, param1_col, param2_col, param3_col, param4_col, param5_col]:
            if col not in df.columns:
                df[col] = np.nan

        tmp = df[["ID", "Case", include_col, op_col, param1_col, param2_col, param3_col,
                  param4_col, param5_col]].copy()
        tmp.columns = ["ID", "Case", "Include", "Operation", "Param1", "Param2", "Param3",
                       "Param4", "Param5"]

        tmp["Scenario"] = include_col if include_col.startswith("S") else f"S{include_col}"

        scenario_rows.append(tmp)

    out = pd.concat(scenario_rows, ignore_index=True)
    ## TODO Dan Modified this code line 14.11.25 ##
    out["Include"] = out["Include"].replace(
        {"Include": True, "Exclude": False, "Yes": True, "No": False, None: False}
           )
    ## end of change ##
           
    for col in ["Param1", "Param2", "Param3", "Param4", "Param5"]:
        out[col] = pd.to_numeric(out[col], errors="coerce").fillna(0)

    return out[["Scenario", "ID", "Case", "Include", "Operation", "Param1", "Param2", "Param3", "Param4", "Param5"]]

def apply_price_adj(df, param1=None, param2=None, param3=None):
    """
    Apply a price change (% change in GM_RATE) and calculate the resulting
    delta on Operating Cash Flow (pre-tax) and RCOP (post-tax).

    Parameters
    ----------
    df : pd.DataFrame
        Subset of data_profile_tall for a single case.
    param1 : float
        % change in price (e.g. 0.10 = +10%, -0.10 = -10%).
    param2 : int
        Start year from which the price change applies.
    param3 : int
        End year until which the price change applies.
    """
    df = df.copy()
    if pd.isna(param1):
        return df

    price_change = float(param1)
    start_year = int(param2) if pd.notna(param2) and param2 != 0 else YEAR_MIN
    end_year = int(param3) if pd.notna(param3) and param3 != 0 else YEAR_MAX

    # --- Get case ID to filter relevant unit rates ---
    case_id = df["ID"].iloc[0]

    # --- 1. Extract GM_RATE for this case ---
    gm_df = (
        unit_rates_tall.loc[
            (unit_rates_tall["ID"] == case_id)
            & (unit_rates_tall["Metric"] == "GM_RATE"),
            ["Year", "Value"]
        ]
        .rename(columns={"Value": "GM_RATE"})
    )

    # --- 2. Extract Production for this case ---
    prod_df = (
        df.loc[df["Metric"] == "Revenue Generating Production", ["Year", "Value"]]
        .rename(columns={"Value": "Revenue Generating Production"})
    )

    # --- 3. Merge both GM_RATE + Production onto df ---
    merged = (
        df.merge(prod_df, on="Year", how="left")
          .merge(gm_df, on="Year", how="left")
    )

    # --- 4. Compute ΔGM only from start_year onward ---
    mask = merged["Year"] >= start_year
    merged.loc[mask, "ΔGM"] = (
        merged.loc[mask, "Revenue Generating Production"].fillna(0)
        * merged.loc[mask, "GM_RATE"].fillna(0)
        * price_change
    )

    # --- 5. Apply ΔGM to Ops Cash and RCOP metrics ---
    target_metrics = [
        "Gross Margin",
        "EBITDA (Underlying)",
        "RCOP (Underlying)",
        "Operating Cashflow",
        "Pre Tax Cash Flow"
    ]
    mask_metric = (
        merged["Metric"].isin(target_metrics)
        & (merged["Year"] >= start_year)
        & (merged["Year"] <= end_year)
    )
    merged.loc[mask_metric, "Value"] = (
        merged.loc[mask_metric, "Value"].fillna(0)
        + merged.loc[mask_metric, "ΔGM"].fillna(0)
    )

    # --- 6. Drop temporary columns ---
    drop_cols = ["Revenue Generating Production", "GM_RATE", "ΔGM"]
    merged = merged.drop(columns=[c for c in drop_cols if c in merged.columns])

    return merged

def apply_CostAdjOpex(df, param1=None, param2=None, param3=None):
    """
    Apply an OPEX change (% change in Cash Costs) and calculate the resulting delta.
    Also adjust related metrics: EBITDA, RCOP, Operating Cashflow, Pre Tax Cash Flow.
    """
    df = df.copy()
    if pd.isna(param1):
        print("⚠️ No param1 provided, returning original DataFrame.")
        return df

    opex_change = float(param1)
    start_year = int(param2) if pd.notna(param2) and param2 != 0 else YEAR_MIN
    end_year = int(param3) if pd.notna(param3) and param3 != 0 else YEAR_MAX

    # --- Debug info ---
    print(f"\n✅ Applying OPEX Adjustment:")
    print(f"Param1 (Change %): {opex_change}, Start Year: {start_year}, End Year: {end_year}")

    # --- Extract Cash Costs ---
    opex_df = (
        df.loc[df["Metric"] == "Cash Costs", ["Year", "Value"]]
        .rename(columns={"Value": "Base_OPEX"})
    )

    merged = df.merge(opex_df, on="Year", how="left")

    # --- Compute ΔOPEX ---
    mask = (merged["Year"] >= start_year) & (merged["Year"] <= end_year)
    merged.loc[mask, "ΔOPEX"] = merged.loc[mask, "Base_OPEX"].fillna(0) * opex_change

    # --- Apply to Cash Costs ---
    mask_cash_cost = (merged["Metric"] == "Cash Costs") & mask
    merged.loc[mask_cash_cost, "Value"] = (
        merged.loc[mask_cash_cost, "Value"].fillna(0) + merged.loc[mask_cash_cost, "ΔOPEX"].fillna(0)
    )

    # --- Apply to related metrics ---
    target_metrics = [
        "EBITDA (Underlying)",
        "RCOP (Underlying)",
        "Operating Cashflow",
        "Pre Tax Cash Flow"
    ]
    mask_target = merged["Metric"].isin(target_metrics) & mask
    merged.loc[mask_target, "Value"] = (
        merged.loc[mask_target, "Value"].fillna(0) + merged.loc[mask_target, "ΔOPEX"].fillna(0)
    )

    # --- Show only changed rows ---
    changed_rows = merged.loc[(mask_cash_cost | mask_target), ["Metric", "Year", "Value", "ΔOPEX"]]
    if changed_rows.empty:
        print("⚠️ No rows were updated. Check metric names and year range.")
    else:
        print("\n🔍 Changed rows after adjustment:")
        print(changed_rows)

    # Drop temp columns
    merged = merged.drop(columns=["Base_OPEX", "ΔOPEX"], errors="ignore")

    return merged

def apply_ProdnAdj(df, param1=None, param2=None, param3=None):
    """
    Production adjustment using BASE production to compute all deltas:
    ΔGM, ΔOPEX, ΔDD&A applied additively (signs already correct in profiles).
    """

    # ---------------------------------------
    # Production metric name declared here
    # ---------------------------------------
    PRODUCTION_METRIC = "Production"

    df = df.copy()
    if pd.isna(param1):
        return df

    change = float(param1)        # e.g. 0.10 or -0.10
    start_year = int(param2) if pd.notna(param2) and param2 != 0 else YEAR_MIN
    end_year   = int(param3) if pd.notna(param3) and param3 != 0 else YEAR_MAX

    case_id = df["ID"].iloc[0]

    # ---------------------------------------
    # 1. GET BASE PRODUCTION
    # ---------------------------------------
    prod_df = (
        df[df["Metric"] == PRODUCTION_METRIC][["Year", "Value"]]
        .rename(columns={"Value": "Base_PRODN"})
    )

    merged = df.merge(prod_df, on="Year", how="left")

    # ---------------------------------------
    # 2. UPDATE PRODUCTION (only for production metric)
    # ---------------------------------------
    mask_year = (merged["Year"] >= start_year) & (merged["Year"] <= end_year)
    mask_prod = (merged["Metric"] == PRODUCTION_METRIC) & mask_year

    merged.loc[mask_prod, "Value"] = (
        merged.loc[mask_prod, "Base_PRODN"].fillna(0) * (1 + change)
    )

    # ---------------------------------------
    # 3. MERGE UNIT RATES
    # ---------------------------------------
    gm_rate  = unit_rates_tall.query("ID == @case_id and Metric == 'GM_RATE'")[["Year","Value"]].rename(columns={"Value":"GM_RATE"})
    opex_rate = unit_rates_tall.query("ID == @case_id and Metric == 'OPEX_RATE'")[["Year","Value"]].rename(columns={"Value":"OPEX_RATE"})
    dda_rate = unit_rates_tall.query("ID == @case_id and Metric == 'DDA_RATE'")[["Year","Value"]].rename(columns={"Value":"DDA_RATE"})
    capex_rate = unit_rates_tall.query("ID == @case_id and Metric == 'CAPEX_RATE'")[["Year","Value"]].rename(columns={"Value":"CAPEX_RATE"})

    merged = (
        merged.merge(gm_rate,  on="Year", how="left")
              .merge(opex_rate, on="Year", how="left")
              .merge(dda_rate,  on="Year", how="left")
              .merge(capex_rate, on="Year", how="left")
    )

    # ---------------------------------------
    # 4. FINANCIAL DELTAS (USE BASE PRODN)
    # ---------------------------------------
    merged["ΔGM"]   = merged["Base_PRODN"].fillna(0) * merged["GM_RATE"].fillna(0) * change
    merged["ΔOPEX"] = merged["Base_PRODN"].fillna(0) * merged["OPEX_RATE"].fillna(0) * change
    merged["ΔDDA"]  = merged["Base_PRODN"].fillna(0) * merged["DDA_RATE"].fillna(0) * change
    merged["ΔCAPEX"]  = merged["Base_PRODN"].fillna(0) * merged["CAPEX_RATE"].fillna(0) * change

    # ---------------------------------------
    # 5. APPLY DELTAS ADDITIVELY TO TARGET METRICS
    # ---------------------------------------

    # Gross Margin
    gm_mask = (merged["Metric"] == "Gross Margin") & mask_year
    merged.loc[gm_mask, "Value"] += merged.loc[gm_mask, "ΔGM"]

    # Opex
    opex_mask = (merged["Metric"] == "Cash Costs") & mask_year
    merged.loc[opex_mask, "Value"] += merged.loc[opex_mask, "ΔOPEX"]

    # Capex
    capex_mask = (merged["Metric"] == "Cash Capex") & mask_year
    merged.loc[capex_mask, "Value"] += merged.loc[capex_mask, "ΔCAPEX"]

    # DD&A
    dda_mask = (merged["Metric"] == "DD&A") & mask_year
    merged.loc[dda_mask, "Value"] += merged.loc[dda_mask, "ΔDDA"]

    # EBITDA
    eb_mask = (merged["Metric"] == "EBITDA (Underlying)") & mask_year
    merged.loc[eb_mask, "Value"] += (
        merged.loc[eb_mask, "ΔGM"] + 
        merged.loc[eb_mask, "ΔOPEX"]
    )

    # RCOP
    rc_mask = (merged["Metric"] == "RCOP (Underlying)") & mask_year
    merged.loc[rc_mask, "Value"] += (
        merged.loc[rc_mask, "ΔGM"] +
        merged.loc[rc_mask, "ΔOPEX"] +
        merged.loc[rc_mask, "ΔDDA"]
    )

    # Operating Cashflow
    ocf_mask = (merged["Metric"] == "Operating Cashflow") & mask_year
    merged.loc[ocf_mask, "Value"] += (
        merged.loc[ocf_mask, "ΔGM"] + 
        merged.loc[ocf_mask, "ΔOPEX"] + 
        merged.loc[ocf_mask, "ΔCAPEX"]
        )

    # Pre Tax CF
    ptcf_mask = (merged["Metric"] == "Pre Tax Cash Flow") & mask_year
    merged.loc[ptcf_mask, "Value"] += (
        merged.loc[ptcf_mask, "ΔGM"] + 
        merged.loc[ptcf_mask, "ΔOPEX"] + 
        merged.loc[ptcf_mask, "ΔCAPEX"]
    )

    # ---------------------------------------
    # 6. CLEAN UP
    # ---------------------------------------
    drop_cols = ["Base_PRODN", "GM_RATE", "OPEX_RATE", "DDA_RATE", "CAPEX_RATE", "ΔGM", "ΔOPEX", "ΔDDA", "ΔCAPEX"]
    merged = merged.drop(columns=[c for c in drop_cols if c in merged.columns], errors="ignore")

    return merged

def apply_marker_price_adj(df, param1=None, param2=None, param3=None, marker="HH_GM_RATE"):
    """
    Apply a marker-specific price change (% change in marker GM rate) and calculate the resulting
    delta on Gross Margin and related metrics.

    Formula:
    ΔGM = Gas Production × Price Change Fraction × Marker GM Rate

    Parameters
    ----------
    df : pd.DataFrame
        Subset of data_profile_tall for a single case.
    param1 : float
        % change in marker price (e.g., 0.10 = +10%, -0.10 = -10%).
    param2 : int
        Start year from which the price change applies.
    param3 : int
        End year until which the price change applies.
    marker : str
        Marker metric name (e.g., "HH_GM_RATE", "NBP_GM_RATE", "JKM_GM_RATE").
    """
    df = df.copy()
    if pd.isna(param1):
        print("⚠️ No param1 provided, returning original DataFrame.")
        return df

    price_change = float(param1)
    start_year = int(param2) if pd.notna(param2) and param2 != 0 else YEAR_MIN
    end_year = int(param3) if pd.notna(param3) and param3 != 0 else YEAR_MAX

    # --- Debug info ---
    print(f"\n✅ Applying {marker} Price Adjustment:")
    print(f"Param1 (Change %): {price_change}, Start Year: {start_year}, End Year: {end_year}")

    case_id = df["ID"].iloc[0]

    # --- Extract Gas Production ---
    gas_df = df.loc[df["Metric"] == "Gas Production", ["Year", "Value"]]
    gas_df = gas_df.rename(columns={"Value": "Gas_Production"})

    # --- Extract Marker Rate ---
    marker_df = unit_rates_tall.loc[
        (unit_rates_tall["ID"] == case_id) & (unit_rates_tall["Metric"] == marker),
        ["Year", "Value"]
    ].rename(columns={"Value": marker})

    # --- Merge data ---
    merged = df.merge(gas_df, on="Year", how="left").merge(marker_df, on="Year", how="left")

    # --- Compute ΔGM ---
    mask = (merged["Year"] >= start_year) & (merged["Year"] <= end_year)
    merged.loc[mask, "ΔGM"] = (
        merged.loc[mask, "Gas_Production"].fillna(0)
        * merged.loc[mask, marker].fillna(0)
        * price_change
    )

    # --- Apply ΔGM to relevant metrics ---
    target_metrics = [
        "Gross Margin",
        "EBITDA (Underlying)",
        "RCOP (Underlying)",
        "Operating Cashflow",
        "Pre Tax Cash Flow"
    ]
    mask_target = merged["Metric"].isin(target_metrics) & mask
    merged.loc[mask_target, "Value"] = (
        merged.loc[mask_target, "Value"].fillna(0) + merged.loc[mask_target, "ΔGM"].fillna(0)
    )

    # --- Show changed rows ---
    changed_rows = merged.loc[mask_target, ["Metric", "Year", "Value", "ΔGM"]]
    if changed_rows.empty:
        print("⚠️ No rows were updated. Check metric names and year range.")
    else:
        print(f"\n🔍 Changed rows after {marker} adjustment:")
        print(changed_rows)

    # --- Drop temp columns ---
    drop_cols = ["Gas_Production", marker, "ΔGM"]
    merged = merged.drop(columns=[c for c in drop_cols if c in merged.columns])

    return merged

    """
    Apply a BRENT price change (% change in BRENT_GM_RATE) and calculate the resulting
    delta on Gross Margin and related metrics.

    Formula:
    ΔGM = Liquids Production × Price Change Fraction × BRENT_GM_RATE

    Parameters
    ----------
    df : pd.DataFrame
        Subset of data_profile_tall for a single case.
    param1 : float
        % change in BRENT price (e.g., 0.10 = +10%, -0.10 = -10%).
    param2 : int
        Start year from which the price change applies.
    param3 : int
        End year until which the price change applies.
    """
    df = df.copy()
    if pd.isna(param1):
        print("⚠️ No param1 provided, returning original DataFrame.")
        return df

    price_change = float(param1)
    start_year = int(param2) if pd.notna(param2) and param2 != 0 else YEAR_MIN
    end_year = int(param3) if pd.notna(param3) and param3 != 0 else YEAR_MAX

    # --- Debug info ---
    print(f"\n✅ Applying BRENT Price Adjustment:")
    print(f"Param1 (Change %): {price_change}, Start Year: {start_year}, End Year: {end_year}")

    case_id = df["ID"].iloc[0]

    # --- Extract Liquids Production ---
    oil_df = df.loc[df["Metric"] == "Liquids Production", ["Year", "Value"]]
    oil_df = oil_df.rename(columns={"Value": "Liquids_Production"})

    # --- Extract BRENT_GM_RATE ---
    brent_df = unit_rates_tall.loc[
        (unit_rates_tall["ID"] == case_id) & (unit_rates_tall["Metric"] == "BRENT_GM_RATE"),
        ["Year", "Value"]
    ].rename(columns={"Value": "BRENT_GM_RATE"})

    # --- Merge data ---
    merged = df.merge(oil_df, on="Year", how="left").merge(brent_df, on="Year", how="left")

    # --- Compute ΔGM ---
    mask = (merged["Year"] >= start_year) & (merged["Year"] <= end_year)
    merged.loc[mask, "ΔGM"] = (
        merged.loc[mask, "Liquids_Production"].fillna(0)
        * merged.loc[mask, "BRENT_GM_RATE"].fillna(0)
        * price_change
    )

    # --- Apply ΔGM to relevant metrics ---
    target_metrics = [
        "Gross Margin",
        "EBITDA (Underlying)",
        "RCOP (Underlying)",
        "Operating Cashflow",
        "Pre Tax Cash Flow"
    ]
    mask_target = merged["Metric"].isin(target_metrics) & mask
    merged.loc[mask_target, "Value"] = (
        merged.loc[mask_target, "Value"].fillna(0) + merged.loc[mask_target, "ΔGM"].fillna(0)
    )

    # --- Show changed rows ---
    changed_rows = merged.loc[mask_target, ["Metric", "Year", "Value", "ΔGM"]]
    if changed_rows.empty:
        print("⚠️ No rows were updated. Check metric names and year range.")
    else:
        print("\n🔍 Changed rows after BRENT adjustment:")
        print(changed_rows)

    # --- Drop temp columns ---
    drop_cols = ["Liquids_Production", "BRENT_GM_RATE", "ΔGM"]
    merged = merged.drop(columns=[c for c in drop_cols if c in merged.columns])

    return merged

def apply_increase(df, param1=None, param2=None, param3=None):
    """
    Proportionally increase Value by (1 + Param1) for all years >= Param2 and <= Param3.
    Param1 = fractional uplift (e.g., 0.1 = +10%)
    Param2 = start year (e.g., 2030)
    Param3 = end year (optional)
    """
    df = df.copy()
    if pd.isna(param1):
        return df
    fraction = float(param1)
    start_year = int(param2) if pd.notna(param2) else df["Year"].min()
    end_year = int(param3) if pd.notna(param3) else df["Year"].max()

    df.loc[(df["Year"] >= start_year) & (df["Year"] <= end_year), "Value"] *= (1 + fraction)
    return df

def apply_dilution(df, param1=None, param2=None, param3=None):
    """
    Reduce Value by (1 - Param1) for all years >= Param2 and <= Param3.
    Param1 = fraction (e.g., 0.25)
    Param2 = start year (e.g., 2030)
    Param3 = end year (optional)
    """
    df = df.copy()
    if pd.isna(param1) or pd.isna(param2):
        return df
    fraction = float(param1)
    start_year = int(param2) if pd.notna(param2) else df["Year"].min()
    end_year = int(param3) if pd.notna(param3) else df["Year"].max()

    df.loc[(df["Year"] >= start_year) & (df["Year"] <= end_year), "Value"] *= (1 - fraction)
    return df
